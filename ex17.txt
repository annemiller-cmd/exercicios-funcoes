#include <stdio.h>
void romano(int n){
  int vals[]    = {1000,900,500,400,100,90,50,40,10,9,5,4,1};
  char *simbs[] = {"M","CM","D","CD","C","XC","L","XL","X","IX","V","IV","I"};
  for(int i=0; i<13; i++){
    while(n >= vals[i]){
      printf("%s", simbs[i]);
        n -= vals[i];
    }
  }
}
int main(){
  int a;
  printf("Digite um número:");
  scanf("%d",&a);
  printf("Em romano: ");
  romano(a);
  return 0;
}
