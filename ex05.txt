#include <stdio.h>
int primo(int n){
  if(n<2){
  return 0;
  }
  for(int i=2;i*i<=n;i++){
    if(n%i==0){
      return 0;
    }
  }
  return 1;
}
int main(){
  int a;
  printf("Digite um número:");
  scanf("%d",&a);
  int r=primo(a);
  if(r==1){
    printf("%d é primo",a);
  }else{
    printf("%d não é primo",a);
  }
  return 0;
}
