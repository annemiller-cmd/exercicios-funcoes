#include <stdio.h>
int triangulo(int a, int b, int c){
  if(a>=b+c||b>=a+c||c>=b+a){
    return 0;
  }
  if(a==b && b==c){
    return 1;
  }
  if(a==b||b==c||c==a){
    return 2;
  }
  return 3;
}
int main(){
  int x,y,z;
  printf("Digite um valor:");
  scanf("%d",&x);
  printf("Digite um valor:");
  scanf("%d",&y);
  printf("Digite um valor:");
  scanf("%d",&z);
  int r=triangulo(x,y,z);
  if(r==0){
    printf("Não é triângulo");
  }
  if(r==1){
    printf("É equilátero");
  }
  if(r==2){
    printf("É isóceles");
  }  
  if(r==3){
    printf("É escaleno");
  }
  return 0;
}
