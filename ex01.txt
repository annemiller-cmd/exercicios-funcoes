#include <stdio.h>
int multiplicacao(int a, int b){
  int resultado=0;
  for(int i=0;i<b;i++){
    resultado+=a;
  }
  return resultado;
}
int main(){
  int n1,n2;
  printf("Digite um número:")
  scanf("%d"",&n1)
  printf("Digite um número:")
  scanf("%d"",&n2);
  int resultado=multiplicacao(n1,n2);
  printf("Multiplicacao:%d",resultado);
  return 0; 
}
