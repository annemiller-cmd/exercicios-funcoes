#include <stdio.h>
void binario(int n){
  int bits[32], tam=0;
  while(n>0){
    bits[tam++]=n%2;
    n/=2;
  }
  for(int i=tam-1;i>=0;i--){
    printf("%d",bits[i]);
  }
}
int main(){
  int n;
  printf("Digite um número:");
  scanf("%d",&n);
  while(n<0){
     printf("Valor inválido, digite novamente:");
    scanf("%d",&n);
  }
  printf("Em binário:");
  binario(n);
  return 0;
}
